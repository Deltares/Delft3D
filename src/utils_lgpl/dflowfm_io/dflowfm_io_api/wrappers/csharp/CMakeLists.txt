set(CSHARP_PROJECT_NAME DFlowFM.IO.NET)
set(CSHARP_SOLUTION_FILE ${CMAKE_CURRENT_SOURCE_DIR}/DFlowFM.IO.NET.slnx)
set(CSHARP_DOTNET_CONFIG $<IF:$<CONFIG:Debug>,Debug,Release>)
set(CSHARP_PLATFORM "Any CPU")

set(DFLOWFM_IO_LIB_STAGE_DIR "${CMAKE_CURRENT_BINARY_DIR}/lib")

add_custom_target(dflowfm_io_csharp_stage_lib ALL
    COMMAND ${CMAKE_COMMAND} -E make_directory "${DFLOWFM_IO_LIB_STAGE_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:dflowfm_io_api>
        "${DFLOWFM_IO_LIB_STAGE_DIR}/$<TARGET_FILE_NAME:dflowfm_io_api>"
    COMMENT "Staging dflowfm_io_api into ${DFLOWFM_IO_LIB_STAGE_DIR}"
    VERBATIM
)
add_dependencies(dflowfm_io_csharp_stage_lib dflowfm_io_api)

file(GENERATE
    OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/CMakeSettings.g.props"
    CONTENT "<!-- Generated by CMake, do not edit -->
<Project>
  <PropertyGroup>
    <OutputPath>${CMAKE_CURRENT_BINARY_DIR}/bin/\$(Configuration)/</OutputPath>
    <BaseIntermediateOutputPath>${CMAKE_CURRENT_BINARY_DIR}/obj/\$(MSBuildProjectName)/</BaseIntermediateOutputPath>
    <DFlowFM_IO_NativeLibrary>${CMAKE_CURRENT_BINARY_DIR}/lib/$<TARGET_FILE_NAME:dflowfm_io_api></DFlowFM_IO_NativeLibrary>
    <DFlowFM_IO_MduJson>${DFLOWFM_IO_MDU_JSON}</DFlowFM_IO_MduJson>
  </PropertyGroup>
</Project>"
)

add_custom_target(dflowfm_io_csharp_build ALL
    COMMAND dotnet build
        ${CSHARP_SOLUTION_FILE}
        --configuration ${CSHARP_DOTNET_CONFIG}
        -p:Platform=${CSHARP_PLATFORM}
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    COMMENT "Building C# solution ${CSHARP_PROJECT_NAME}"
)
add_dependencies(dflowfm_io_csharp_build dflowfm_io_api dflowfm_io_csharp_stage_lib)

if(DFLOWFM_IO_BUILD_WRAPPER_PACKAGES)
    set(_dflowfm_io_csharp_pack_in_all ALL)
else()
    set(_dflowfm_io_csharp_pack_in_all "")
endif()

add_custom_target(dflowfm_io_csharp_pack ${_dflowfm_io_csharp_pack_in_all}
    COMMAND dotnet pack
        ${CSHARP_SOLUTION_FILE}
        --configuration ${CSHARP_DOTNET_CONFIG}
        -p:Platform=${CSHARP_PLATFORM}
        --no-build
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    COMMENT "Creating NuGet package for C# solution ${CSHARP_PROJECT_NAME} in ${CSHARP_OUTPUT_DIR}"
)
add_dependencies(dflowfm_io_csharp_pack dflowfm_io_csharp_build)

add_test(NAME dflowfm_io_csharp_test
    COMMAND dotnet test
        ${CSHARP_SOLUTION_FILE}
        --configuration ${CSHARP_DOTNET_CONFIG}
        -p:Platform=${CSHARP_PLATFORM}
        --no-build
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
)
